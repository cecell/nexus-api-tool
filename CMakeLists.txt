cmake_minimum_required(VERSION 3.21)

# One self-contained exe: static CRT, so it runs without the Visual C++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(VCPKG_TARGET_TRIPLET "x64-windows-static" CACHE STRING "")

# The one place for the version: the exe and the release workflow both read it from here.
project(nexus_api_tool VERSION 0.5.1 LANGUAGES CXX)

find_package(nlohmann_json CONFIG REQUIRED)

add_executable(nexus-api-tool src/main.cpp)
target_compile_features(nexus-api-tool PRIVATE cxx_std_17)
target_compile_definitions(nexus-api-tool PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS APP_VERSION="${PROJECT_VERSION}")
target_compile_options(nexus-api-tool PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(nexus-api-tool PRIVATE nlohmann_json::nlohmann_json winhttp advapi32 bcrypt crypt32)

# Command-line checks that need neither the network nor an API key.
enable_testing()
function(add_cli_test name expected)
    add_test(NAME ${name} COMMAND nexus-api-tool ${ARGN})
    set_tests_properties(${name} PROPERTIES PASS_REGULAR_EXPRESSION "${expected}")
endfunction()
add_cli_test(version "nexus-api-tool ${PROJECT_VERSION}" --version)
add_cli_test(help "USAGE" --help)
add_cli_test(unknown_command "unknown command nope" nope 1)
add_cli_test(bad_mod_id "the mod id must be a number" files abc)
add_cli_test(missing_dest "dl-file needs --dest" dl-file 790 937)
add_cli_test(limit_too_high "--limit must be 1 to 200" dependents 266 --limit 201)
add_cli_test(limit_zero "--limit must be 1 to 200" dependents 266 --limit 0)
add_cli_test(limit_wrong_command "only apply to reqs, dependents and list-mods" files 266 --limit 5)
add_cli_test(set_desc_needs_file "set-desc needs --file" set-desc 266)
add_cli_test(save_desc_needs_dest "save-desc needs --dest" save-desc 266)
add_cli_test(upload_needs_name "upload-file needs --name and --file-version" upload-file 266 --file x.7z)
add_cli_test(option_wrong_command "--file doesn't apply to desc" desc 266 --file x)
add_cli_test(list_mods_needs_user "list-mods needs a user name" list-mods)
add_cli_test(permissions_bad_address "must be a number or a mod page" permissions https://example.com/x)
add_cli_test(permissions_saved_page "You must get permission from me" permissions 1 --html ${CMAKE_CURRENT_SOURCE_DIR}/tests/permissions_sample.html)
add_cli_test(help_rate_limits "Cloudflare may impose a limit of 4/hour" --help)

# Description projects: offline, in a copy of tests/desc_root in the build folder.
set(DESC_ROOT ${CMAKE_CURRENT_BINARY_DIR}/desc_root)
add_test(NAME desc_setup COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/tests/desc_root ${DESC_ROOT})
set_tests_properties(desc_setup PROPERTIES FIXTURES_SETUP desc)
function(add_desc_test name expected)
    add_cli_test(${name} "${expected}" ${ARGN})
    set_tests_properties(${name} PROPERTIES FIXTURES_REQUIRED desc)
endfunction()
add_cli_test(desc_split "\"part\": \"intro.install\"" desc-split --file ${CMAKE_CURRENT_SOURCE_DIR}/tests/desc_page.bbcode)
add_desc_test(desc_templates "Needs the author variable" desc-templates --root ${DESC_ROOT})
add_desc_test(desc_init "\"part\": \"credits\"" desc-init 1 --root ${DESC_ROOT} --from-file ${CMAKE_CURRENT_SOURCE_DIR}/tests/desc_page.bbcode --force)
add_desc_test(desc_build_page "\"round_trip\": true" desc-build 1 --root ${DESC_ROOT} --out ${CMAKE_CURRENT_BINARY_DIR}/desc_page_built.bbcode)
set_tests_properties(desc_build_page PROPERTIES DEPENDS desc_init)
add_test(NAME desc_page_identical COMMAND ${CMAKE_COMMAND} -E compare_files --ignore-eol ${CMAKE_CURRENT_BINARY_DIR}/desc_page_built.bbcode ${CMAKE_CURRENT_SOURCE_DIR}/tests/desc_page.bbcode)
set_tests_properties(desc_page_identical PROPERTIES DEPENDS desc_build_page)
add_desc_test(desc_new "\"template\": \"basic\"" desc-new 3 --root ${DESC_ROOT} --template basic --offline --force)
add_desc_test(desc_build_template "\"round_trip\": true" desc-build 3 --root ${DESC_ROOT} --out ${CMAKE_CURRENT_BINARY_DIR}/desc_template_built.bbcode)
set_tests_properties(desc_build_template PROPERTIES DEPENDS desc_new)
add_test(NAME desc_template_identical COMMAND ${CMAKE_COMMAND} -E compare_files --ignore-eol ${CMAKE_CURRENT_BINARY_DIR}/desc_template_built.bbcode ${CMAKE_CURRENT_SOURCE_DIR}/tests/desc_expected_template.bbcode)
set_tests_properties(desc_template_identical PROPERTIES DEPENDS desc_build_template)
add_desc_test(desc_new_missing_var "\"template\": \"byauthor\"" desc-new 4 --root ${DESC_ROOT} --template byauthor --offline --force)
add_desc_test(desc_build_missing_var "not filled in: author" desc-build 4 --root ${DESC_ROOT})
set_tests_properties(desc_build_missing_var PROPERTIES DEPENDS desc_new_missing_var)
add_desc_test(desc_set_part "\"part\": \"intro.install\"" desc-set 3 intro.install --root ${DESC_ROOT} --text "New install text.")
set_tests_properties(desc_set_part PROPERTIES DEPENDS desc_template_identical)
add_desc_test(desc_set_unknown "the parts are preamble, intro, intro.install, credits" desc-set 3 nope --root ${DESC_ROOT} --text x)
set_tests_properties(desc_set_unknown PROPERTIES DEPENDS desc_new)
add_cli_test(desc_new_needs_template "desc-new needs --template" desc-new 5 --root .)
add_cli_test(desc_set_needs_text "desc-set takes --file <path> or --text <text>" desc-set 5 intro --root .)
